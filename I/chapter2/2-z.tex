\section*{3}

\begin{itemize}
  \item (1)

        (i)

        $\lim\limits_{n \to \infty} a_n = a$，由极限定义可知，
        对任意$\epsilon > 0$，存在$N_1 > 0$，当$n > N_1$时，有
        \begin{align*}
          |a_n - a| < \epsilon
        \end{align*}
        于是
        \begin{align*}
          |\frac{a_1 + a_2 + \cdots + a_n}{n} - a|
           & = |\frac{a_1}{n} + \frac{a_2}{n} + \cdots + \frac{a_n}{n} - a|                      \\
           & = \frac{1}{n}|(a_1 - a) + (a_2 - a) + \cdots + (a_n - a)|                           \\
           & \leq \frac{1}{n} |a_1 - a| + \frac{1}{n} |a_2 - a| + \cdots + \frac{1}{n} |a_n - a|
        \end{align*}
        取$M = \max\{|a_1 - a|, |a_2 - a|, \cdots, |a_{n-1} - a|\}$，于是
        \begin{align*}
          \frac{1}{n} |a_1 - a| + \frac{1}{n} |a_2 - a| + \cdots + \frac{1}{n} |a_n - a|
           & \leq \frac{N_1M}{n} + \frac{n - N_1}{n}\epsilon \\
           & \leq \frac{N_1M}{n} + \epsilon
        \end{align*}
        由于
        \begin{align*}
          \lim\limits_{n \to \infty} \frac{N_1M}{n} = 0
        \end{align*}
        所以，存在$N_2 > 0$，当$n > N_2$时，有
        \begin{align*}
          |\frac{N_1M}{n}| < \epsilon
        \end{align*}
        综上，取$N = max\{N_1, N_2\}$，当$n > N$时，有
        \begin{align*}
          |\frac{a_1 + a_2 + \cdots + a_n}{n} - a|
           & \leq \frac{N_1M}{n} + \frac{n - N_1}{n}\epsilon \\
           & \leq \epsilon + \epsilon                        \\
           & = 2\epsilon
        \end{align*}
        所以，$\lim\limits_{n \to \infty} \frac{a_1 + a_2 + \cdots + a_n}{n} = a$。

        (ii)

        逆向不成立，比如
        \begin{align*}
          a_n = (-1)^n
        \end{align*}
        此时
        \begin{align*}
          \lim\limits_{n \to \infty} \frac{a_1 + a_2 + \cdots + a_n}{n} = 0
        \end{align*}
        而
        \begin{align*}
          \lim\limits_{n \to \infty} a_n
        \end{align*}
        不存在。

  \item (2)

        因为$a_n > 0, n = 1, 2, \cdots, n$，我们有
        \begin{align*}
          \frac{a_1 + a_2 + \cdots + a_n}{n} \geq \sqrt[n]{a_1 a_2 \cdots a_n} \geq \frac{n}{\frac{1}{a_1} + \frac{1}{a_2} + \cdots + \frac{1}{a_n}}
        \end{align*}

        由(1)，我们有
        \begin{align*}
          \lim\limits_{n \to \infty} \frac{a_1 + a_2 + \cdots + a_n}{n} = a
        \end{align*}

        又因为
        \begin{align*}
          \lim \limits_{n \to \infty} \frac{1}{a_n} = \frac{1}{a}
        \end{align*}
        利用(1)，我们有
        \begin{align*}
          \lim\limits_{n \to \infty} \frac{\frac{1}{a_1} + \frac{1}{a_2} + \cdots + \frac{1}{a_n}}{n}
          = \frac{1}{a}
        \end{align*}
        所以
        \begin{align*}
          \lim\limits_{n \to \infty} \frac{n}{\frac{1}{a_1} + \frac{1}{a_2} + \cdots + \frac{1}{a_n}} = a
        \end{align*}

        由保不等式性可知
        \begin{align*}
          \lim\limits_{n \to \infty} \sqrt[n]{a_1 a_2 \cdots a_n} = a
        \end{align*}
\end{itemize}
