\section*{1}

\begin{itemize}
  \item (1)

        令$y = x - \frac{\pi}{2}$，$x \to \pi/2$时，$y \to 0$且$t \neq 0$，
        于是，我们有
        \begin{align*}
          \lim\limits_{x \to \frac{\pi}{2}} \frac{cosx}{x - \frac{\pi}{2}}
          & = \lim\limits_{y \to 0} \frac{cos(y + \frac{\pi}{2})}{y} \\
          & = \lim\limits_{y \to 0} \frac{-sin(y)}{y} \\
          & = -1
        \end{align*}

  \item (8)

        \begin{align*}
          \lim\limits_{x \to a} \frac{sin^2(x) - sin^2(a)}{x - a}
           & = \lim\limits_{x \to a} \frac{[sin(x) - sin(a)][sin(x) + sin(a)]}{x - a}                                \\
           & = \lim\limits_{x \to a} \frac{2cos(\frac{x+a}{2})sin(\frac{x-a}{2})[sin(x) + sin(a)]}{x - a}            \\
           & = \lim\limits_{x \to a} [sin(x) + sin(a)] cos(\frac{x+a}{2}) \frac{sin(\frac{x-a}{2})}{\frac{x - a}{2}} \\
           & = 2sin(a) \cdot cos(a) \cdot 1                                                                          \\
           & = sin(2a)
        \end{align*}
\end{itemize}

        \begin{align*}
          \lim\limits_{x \to 0} \left(\frac{1 + x}{1 - x}\right)^{\frac{1}{x}}
           & = \lim\limits_{x \to 0} \left(1 + \frac{2x}{1 - x}\right)^{\frac{1}{x}}                       \\

        \end{align*}

\section*{3}

\begin{zremark}
  这种极限套极限的格式，按照运算的优先级进行计算。本题中，先计算大括号里的极限，把其中的$x$看做定值。
\end{zremark}

我们有
\begin{align*}
  sinx & = 2 cos\frac{x}{2} sin\frac{x}{2}                                             \\
       & = \cdots                                                                      \\
       & = 2^n cos\frac{x}{2} cos\frac{x}{2^2} \cdots cos\frac{x}{2^n}sin\frac{x}{2^n}
\end{align*}
于是，
\begin{align*}
  \lim_{n\to\infty} cosx cos\frac{x}{2} + \cdots + cos\frac{x}{2^n}
   & = \lim_{n\to\infty} \frac{cosx sinx}{2^n sin\frac{x}{2^n}}                    \\
   & = \lim_{n\to\infty} \frac{cosx sinx}{x}\frac{\frac{x}{2^n}}{sin\frac{x}{2^n}} \\
   & = \frac{cosx sinx}{x} \cdot 1                                                 \\
   & = \frac{cosx sinx}{x}
\end{align*}

所以
\begin{align*}
  \lim\limits_{x \to 0} \frac{cosx sinx}{x} = 1
\end{align*}
