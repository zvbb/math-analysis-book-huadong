\section*{2}

\begin{align*}
  \frac{d}{dx} f(x^2)
   & = f'(x) \cdot 2x
\end{align*}
和
\begin{align*}
  \frac{d^2}{dx^2} f^2(x)
   & = C_2^0 f''(x)f(x) + C_2^1 f'(x)f'(x) + C_2^2 f(x) f''(x)
\end{align*}
所以在$x=1$处，两个导数都等于$0$。

          \frac{dy}{dx}
           & = \frac{3a sin^2(t) cos(t)}{-3acos^2(t)sin(t)} \\

          & = \frac{1}{3acos^4(t)sin(t)}
        \end{align*}
