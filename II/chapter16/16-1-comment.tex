\begin{zremark}
  证明：平面三角形不等式，设
  \begin{align*}
    P_1 & = (x_1, y_1) \\
    P_2 & = (x_2, y_2) \\
    P_3 & = (x_3, y_3)
  \end{align*}
  可得
  \begin{align*}
    \rho(P_1, P_2) \leq \rho(P_1, P_3) + \rho(P_2, P_3)
  \end{align*}
\end{zremark}
\textbf{证明：}
我们有
\begin{align*}
   & \rho(P_1, P_2)                                                  \\
   & = \sqrt{(x_2 - x_1)^2 + (y_2 - y_1)^2}                          \\
   & = \sqrt{(x_2 - x_3 + x_3 - x_1)^2 + (y_2 - y_3 + y_3 - y_1)^2}  \\
   & = \sqrt{(x_2 - x_3)^2 + (x_3 - x_1)^2 + 2(x_2 - x_3)(x_3 - x_1)
  + (y_2 - y_3)^2 + (y_3 - y_1)^2 + 2 (y_2 - y_3)(y_3 - y_1)}        \\
\end{align*}
和
\begin{align*}
  \rho(P_1, P_3) & = \sqrt{(x_3 - x_1)^2 + (y_3 - y_1)^2} \\
  \rho(P_2, P_3) & = \sqrt{(x_2 - x_3)^2 + (y_2 - y_3)^2}
\end{align*}
于是
\begin{align*}
   & [\rho(P_1, P_3) + \rho(P_2, P_3)]^2 - \rho^2(P_1, P_2)                       \\
   & = 2 \sqrt{(x_3 - x_1)^2 + (y_3 - y_1)^2}\sqrt{(x_2 - x_3)^2 + (y_2 - y_3)^2}
  - 2(x_2 - x_3)(x_3 - x_1) - 2 (y_2 - y_3)(y_3 - y_1)                            \\
   & = 2 \sqrt{[(x_3 - x_1)^2 + (y_3 - y_1)^2][(x_2 - x_3)^2 + (y_2 - y_3)^2]}
  - 2(x_2 - x_3)(x_3 - x_1) - 2 (y_2 - y_3)(y_3 - y_1)
\end{align*}
由柯西-施瓦茨不等式（证明base.tex中有证明）
\begin{align*}
  [(x_3 - x_1)^2 + (y_3 - y_1)^2][(x_2 - x_3)^2 + (y_2 - y_3)^2]
  \geq [(x_3 - x_1)(x_2 - x_3) + (y_3 - y_1)(y_2 - y_3)]^2
\end{align*}
代入上式可得
\begin{align*}
  [\rho(P_1, P_3) + \rho(P_2, P_3)]^2 - \rho^2(P_1, P_2) \geq 0
\end{align*}
命题得证。

\begin{zremark}
  闭域$D$被开域$S$表示为
  \begin{align*}
    D = S \cup \partial S
  \end{align*}
  则$\partial D = \partial S$.
\end{zremark}
\textbf{证明：}
反证法，假设存在$P_0 \in \partial D, P_0 \notin \partial S$。

由界点的定义，$P_0$可能属于$D$，也可能不属于$D$.

(i)如果$P_0 \in D$，即$P_0 \in S$，由于$S$是开域，则$P_0$是$S$的内点，
由内点的定义可知，$P_0$也是$D$的内点，出现矛盾。

(ii)$P_0 \notin D$，由界点的定义，$P_0$的任意邻域$U(P_0)$都含有$D$
的点，又因为$P_0 \notin D$，综上可得$P_0$
是$D$的聚点，由习题4可知，$D$也是闭集，则有$P_0 \in D$，
出现矛盾。

综上，假设不成立，$\partial D = \partial S$得证。

\begin{zremark}
  证明：定理$16.3'$
\end{zremark}
\textbf{证明：}
有界无限点集和有界无限点列的主要区别在于：
\begin{itemize}
  \item 有界无限点集：包含无限多个互不相同的点，且所有点都位于某个有界区域内；
  \item 有界无限点列：允许重复元素
\end{itemize}
把有界无限点列$\{P_n\}$看做点集$E$：

(i)如果$\{P_n\}$中只有有限个不同的点，则必有至少一个点在点列中出现无限次，
任选其中一个点（设为$P_0$），
令${n_k}$为所有满足$P_{n_k} = P_0$的下标，
并按递增顺序排列（即$n_1 < n_2 < n_3 < \cdots$）。
这样得到的子列${P_{n_k}}$是一个常数序列（所有项均为$P_0$），
因此显然收敛于$P_0$。

(ii)如果$\{P_n\}$中有无限个不同的点，则$E$是有界无限点集，
由聚点定理，$E$中至少有一个聚点，设为$P_0$，
由聚点的定义，对于任意的$\epsilon > 0$，存在$E$中的点$P \neq P_0$，
使得
\begin{align*}
  \rho(P, P_0) < \epsilon
\end{align*}
现在，我们利用这个定义构造一个收敛于$P_0$的子列$P_{n_k}$。
具体构造方式如下：

取$\epsilon = 1$，由聚点定义存在$P_{n_1} \in E$，使得$\rho(P_{n_1}, P_0) < 1$；

取$\epsilon = \frac{1}{2}$，由聚点定义
$U(P_0; \frac{1}{2})$邻域中有无数个点，存在点$P_{n_2}$（$n_2 > n_1$）
使得$\rho(P_{n_2}, P_0) < \frac{1}{2}$；

$\vdots$

取$\epsilon = \frac{1}{l}$，由聚点定义
$U(P_0; \frac{1}{l})$邻域中有无数个点，存在点$P_{n_l}$（$n_l > n_{l - 1}$）
使得$\rho(P_{n_l}, P_0) < \frac{1}{l}$；

将这个过程无限重复下去，就得到一个收敛于$P_0$的子列$P_{n_k}$。
