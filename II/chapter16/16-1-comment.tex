\begin{zremark}
  证明：平面三角形不等式，设
  \begin{align*}
    P_1 & = (x_1, y_1) \\
    P_2 & = (x_2, y_2) \\
    P_3 & = (x_3, y_3)
  \end{align*}
  可得
  \begin{align*}
    \rho(P_1, P_2) \leq \rho(P_1, P_3) + \rho(P_2, P_3)
  \end{align*}
\end{zremark}
\textbf{证明：}
我们有
\begin{align*}
   & \rho(P_1, P_2)                                                  \\
   & = \sqrt{(x_2 - x_1)^2 + (y_2 - y_1)^2}                          \\
   & = \sqrt{(x_2 - x_3 + x_3 - x_1)^2 + (y_2 - y_3 + y_3 - y_1)^2}  \\
   & = \sqrt{(x_2 - x_3)^2 + (x_3 - x_1)^2 + 2(x_2 - x_3)(x_3 - x_1)
  + (y_2 - y_3)^2 + (y_3 - y_1)^2 + 2 (y_2 - y_3)(y_3 - y_1)}        \\
\end{align*}
和
\begin{align*}
  \rho(P_1, P_3) & = \sqrt{(x_3 - x_1)^2 + (y_3 - y_1)^2} \\
  \rho(P_2, P_3) & = \sqrt{(x_2 - x_3)^2 + (y_2 - y_3)^2}
\end{align*}
于是
\begin{align*}
   & [\rho(P_1, P_3) + \rho(P_2, P_3)]^2 - \rho^2(P_1, P_2)                       \\
   & = 2 \sqrt{(x_3 - x_1)^2 + (y_3 - y_1)^2}\sqrt{(x_2 - x_3)^2 + (y_2 - y_3)^2}
  - 2(x_2 - x_3)(x_3 - x_1) - 2 (y_2 - y_3)(y_3 - y_1)                            \\
   & = 2 \sqrt{[(x_3 - x_1)^2 + (y_3 - y_1)^2][(x_2 - x_3)^2 + (y_2 - y_3)^2]}
  - 2(x_2 - x_3)(x_3 - x_1) - 2 (y_2 - y_3)(y_3 - y_1)
\end{align*}
由柯西-施瓦茨不等式（证明base.tex中有证明）
\begin{align*}
  [(x_3 - x_1)^2 + (y_3 - y_1)^2][(x_2 - x_3)^2 + (y_2 - y_3)^2]
  \geq [(x_3 - x_1)(x_2 - x_3) + (y_3 - y_1)(y_2 - y_3)]^2
\end{align*}
代入上式可得
\begin{align*}
  [\rho(P_1, P_3) + \rho(P_2, P_3)]^2 - \rho^2(P_1, P_2) \geq 0
\end{align*}
命题得证。
