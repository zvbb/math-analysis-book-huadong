\begin{zremark}
  多元函数的偏导数的四则运算：

  设$f, g : U \subset R^n \to \mathbb{R}$，且在点$x(x_1, x_2, \cdots, x_n) \in U$可微。
  则我们有
  \begin{align*}
    \frac{\partial}{\partial x_i} (f \pm g) (x)                & = \frac{\partial f}{\partial x_i}(x) \pm \frac{\partial g}{\partial x_i}(x)                        \\
    \frac{\partial}{\partial x_i} (f g) (x)                    & = f(x)\frac{\partial g}{\partial x_i}(x) + g(x)\frac{\partial f}{\partial x_i}(x)                  \\
    \frac{\partial}{\partial x_i} \left(\frac{f}{g}\right) (x) & = \frac{g(x)\frac{\partial f}{\partial x_i}(x) - f(x)\frac{\partial g}{\partial x_i}(x)}{[g(x)]^2}
  \end{align*}
\end{zremark}
\textbf{证明：}

以乘法为例
\begin{align*}
  \frac{\partial}{\partial x_1} (f g) (x)
   & = \lim\limits_{h \to 0} \frac{(f g) (x_1 + h, \cdots, x_n) - (f g) (x_1, \cdots, x_n)}{h}                                     \\
   & = \lim\limits_{h \to 0} \frac{f(x_1 + h, \cdots, x_n)g(x_1 + h, \cdots, x_n) - f (x_1, \cdots, x_n) g (x_1, \cdots, x_n)}{h}  \\
   & = \lim\limits_{h \to 0} \frac{f(x_1 + h, \cdots, x_n)g(x_1 + h, \cdots, x_n) - f(x_1, \cdots, x_n)g(x_1 + h, \cdots, x_n)}{h} \\
   & \quad + \frac{f(x_1, \cdots, x_n)g(x_1 + h, \cdots, x_n) - f (x_1, \cdots, x_n) g (x_1, \cdots, x_n)}{h}                      \\
   & = \lim\limits_{h \to 0} \frac{f(x_1 + h, \cdots, x_n) - f(x_1, \cdots, x_n)}{h} g(x_1 + h, \cdots, x_n)                       \\
   & \quad + \frac{g(x_1 + h, \cdots, x_n) - g (x_1, \cdots, x_n)}{h} f (x_1, \cdots, x_n)                                         \\
   & = \lim\limits_{h \to 0} \frac{f(x_1 + h, \cdots, x_n) - f(x_1, \cdots, x_n)}{h} g(x_1 + h, \cdots, x_n)                       \\
   & \quad + \lim\limits_{h \to 0} \frac{g(x_1 + h, \cdots, x_n) - g (x_1, \cdots, x_n)}{h} f (x_1, \cdots, x_n)
\end{align*}
因为$f,g$在$x$处可微，所以在$x$处连续，于是可得
\begin{align*}
   & \lim\limits_{h \to 0} \frac{f(x_1 + h, \cdots, x_n) - f(x_1, \cdots, x_n)}{h} g(x_1 + h, \cdots, x_n)       \\
   & \quad + \lim\limits_{h \to 0} \frac{g(x_1 + h, \cdots, x_n) - g (x_1, \cdots, x_n)}{h} f (x_1, \cdots, x_n) \\
   & = g(x)\frac{\partial f}{\partial x_i}(x) + f(x)\frac{\partial g}{\partial x_i}(x)
\end{align*}

\begin{zremark}
  命题

  设$z = f(x, y), x = \varphi (s, t), y = \psi (s, t)$，
  若函数$f, \varphi, \psi$都具有连续的二阶偏导数，则$z$对
  $s, t$同样存在二阶连续偏导数。
\end{zremark}

\textbf{证明：}

有定理17.5可知，复合函数
\begin{align*}
  z = F(s, t) = f(\varphi (s, t), \psi (s, t))
\end{align*}
是可微的，且我们有
\begin{align*}
  F_s = \frac{\partial z}{\partial x} \frac{\partial x}{\partial s} + \frac{\partial z}{\partial y} \frac{\partial y}{\partial s} \\
  F_t = \frac{\partial z}{\partial x} \frac{\partial x}{\partial t} + \frac{\partial z}{\partial y} \frac{\partial y}{\partial t}
\end{align*}
接下来证明$F_s$（$F_t$类似）是可微，从而偏导数。考虑
\begin{align*}
  F_s
   & = \frac{\partial z}{\partial x} \frac{\partial x}{\partial s} + \frac{\partial z}{\partial y} \frac{\partial y}{\partial s}       \\
   & = f_x \frac{\partial x}{\partial s} + f_y \frac{\partial y}{\partial s}                                                           \\
   & = f_x(x, y) \frac{\partial x}{\partial s} + f_y(x, y) \frac{\partial y}{\partial s}                                               \\
   & = f_x(\varphi (s, t), \psi (s, t)) \frac{\partial x}{\partial s} + f_y(\varphi (s, t), \psi (s, t)) \frac{\partial y}{\partial s} \\
\end{align*}
以上可以得到$f_x(\varphi (s, t), \psi (s, t))$是$f_x(x, y), x = \varphi (s, t), y = \psi (s, t)$的复合，
再次利用17.5可知，$f_x(\varphi (s, t), \psi (s, t))$可微，且偏导数满足链式法则；

同理可得$f_y(\varphi (s, t), \psi (s, t))$可微，且偏导数满足链式法则；
然后利用多元函数的偏导数的四则运算，可得$F_{ss}$（具体结果P.124），
由于$F_{ss}$中的每一项都是连续的，故$F_{ss}$连续。

\begin{zremark}
  $f_{xy}(x, y)$在点$(x_0, y_0)$连续，则存在$(x_0, y_0)$的领域$U$，
  对任意$(x, y) \in U$，$f$有定义。
\end{zremark}
\textbf{证明：}

按照偏导数$f_{xy}$的定义:
\begin{align*}
  f_{xy}(x_0, y_0) = \lim\limits_{\Delta y \to 0} \frac{f_x(x_0, y_0 + \Delta y) - f_x(x_0, y_0)}{\Delta y}
\end{align*}
可知偏导数$f_x$在$y_0$的某个领域内有定义。
同理，按照$f_x$的定义：
\begin{align*}
  f_x(x_0, y_0 + \Delta y) = \lim\limits_{\Delta x \to 0} \frac{f(x_0 + \Delta x, y_0 + \Delta y) - f(x_0, y_0 + \Delta y)}{\Delta x}
\end{align*}
可知对每个充分小的$\Delta y$，$f$在$(*, y_0 + \Delta y)$在$x_0$的某个领域内有定义；

综上，$f$在$(x_0, y_0)$的某个领域内有定义；

\begin{zremark}
  $f_{xy}(x, y)$在点$(x_0, y_0)$的邻域上有定义，且在$(x_0, y_0)$处连续，则存在$(x_0, y_0)$的领域$U$，
  对任意$(x, y) \in U$，$f_x(x, y)$存在。
\end{zremark}
\textbf{证明：}

由题设可知，存在$(x_0, y_0)$的领域$U$，使得$f_{xy}$在$U$内每一个点都有定义；

对任意$(x_1, y_1) \in U$，由偏导数的定义，我们有
\begin{align*}
  f_{xy} (x_1, y_1) = \lim\limits_{\Delta y \to 0} \frac{f_x(x_1, y_1 + \Delta y) - f_x(x_1, y_1)}{\Delta y}
\end{align*}
所以$f_x(x_1, y_1)$存在，命题成立。

\begin{zremark}
  定理17.9（泰勒定理）证明过程中：

  \begin{align*}
    \varPhi^{(m)}(t) = \left( h\frac{\partial}{\partial x} + k\frac{\partial}{\partial y} \right)^m
    f(x_0 + th, y_0 + tk) \quad (m = 1, 2, \cdots, n + 1)
  \end{align*}
\end{zremark}

\textbf{证明：}

归纳法证明，$m = 1$，使用链式法则可知等式成立。

归纳假设$m = n$时，等式成立，接下来证明$m = n + 1$的情况。

我们有
\begin{align*}
  \varPhi^{(n + 1)}
   & = \frac{d}{dt} \varPhi^{(n)}                                                                      \\
   & = \frac{d}{dt} \left[\left( h\frac{\partial}{\partial x} + k\frac{\partial}{\partial y} \right)^n
    f(x_0 + th, y_0 + tk)\right]
\end{align*}
令
\begin{align*}
  g(x, y) = \left( h\frac{\partial}{\partial x} + k\frac{\partial}{\partial y} \right)^n f(x, y)
\end{align*}
题设保证$g(x, y)$存在连续的偏导数，则由链式法则，有
\begin{align*}
  \varPhi^{(n + 1)}
   & = \frac{d}{dt} g(x_0 + th, y_0 + tk)                                                                                                                                             \\
   & = \left( h\frac{\partial}{\partial x} + k\frac{\partial}{\partial y} \right) g(x_0 + th, y_0 + tk)                                                                               \\
   & = \left( h\frac{\partial}{\partial x} + k\frac{\partial}{\partial y} \right) \left( h\frac{\partial}{\partial x} + k\frac{\partial}{\partial y} \right)^n f(x_0 + th, y_0 + tk) \\
   & = \left( h\frac{\partial}{\partial x} + k\frac{\partial}{\partial y} \right)^{n + 1} f(x_0 + th, y_0 + tk)
\end{align*}

\begin{zremark}
  证明：定理17.10（极值必要条件）
\end{zremark}
\textbf{证明：}
以$f_x(x_0, y_0) = 0$为例。

反证法，假设$f_x(x_0, y_0) \neq 0$，不妨设$f_x > 0$，
由偏导数的定义可知
\begin{align*}
  \lim\limits_{\Delta x \to 0} \frac{f(x_0 + \Delta x, y_0) - f(x_0, y_0)}{\Delta x}
  = f_x(x_0, y_0) > 0
\end{align*}
由保号性可知，存在$\delta > 0$，$0 < \Delta x < \delta$时，有
\begin{align*}
  \frac{f(x_0 + \Delta x, y_0) - f(x_0, y_0)}{\Delta x} & > 0           \\
  f(x_0 + \Delta x, y_0) - f(x_0, y_0)                  & > 0           \\
  f(x_0 + \Delta x, y_0)                                & > f(x_0, y_0)
\end{align*}
这与$f(x_0, y_0)$是极值点矛盾。
